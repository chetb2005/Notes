### To deploy nginx
mkdir project && cd project

nano project-deployment.yaml

###############################

apiVersion: apps/v1
kind: Deployment

metadata:
  name: project-deployment

spec:
  replicas: 1

  selector:
    matchLabels:
      app: custom

  template:
    metadata:
      labels:
        app: custom

    spec:
      nodeSelector:
        kubernetes.io/hostname: k8s-node2

      containers:
        - name: itiss
          image: nasawebtech/itiss:latest
          ports:
            - containerPort: 80
##################################################################



Deploy

sudo k3s kubectl apply -f project-deployment.yaml

################################################################
Create the SErvice - so that you can access it from outside the cluster

nano project-service.yaml

#################

apiVersion: v1
kind: Service

metadata:
  name: project-service

spec:
  type: NodePort

  selector:
    app: custom

  ports:
    - port: 80
      targetPort: 80
      nodePort: 30080
      protocol: TCP
############


Apply the service

sudo k3s kubectl apply -f project-service.yaml


# Access the site


http://3.107.45.176:30080



# change the replicas

sudo k3s kubectl scale deployment project-deployment --replicas=2

# Remove Service
sudo k3s kubectl delete -f project-service.yaml

#Remove deployment
sudo k3s kubectl delete -f project-deployment.yaml
