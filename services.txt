services:
  nginx:
    image: nginx
    container_name: rproxy
    ports:
    - "80:80"
    volumes:
    - ./rproxy/nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - web1
      - web2
    restart: unless-stopped
    networks:
      - web-network

  web1:
    image: ubuntu/apache2
    container_name: web1
    ports:
      - "80"
    volumes:
      - ./web1:/var/www/html:ro
    restart: unless-stopped
    networks:
      - web-network


  web2:
    image: ubuntu/apache2
    container_name: web2
    ports:
      - "80"
    volumes:
      - ./web2:/var/www/html:ro
    restart: unless-stopped
    networks:
      - web-network

networks:
  web-network:
    driver: bridge
